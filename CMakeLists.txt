cmake_minimum_required(VERSION 3.24)
project(Crucible VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Optimised unless asked otherwise: timings and long runs are only meaningful optimised.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(CRUCIBLE_BUILD_APP "Build the native desktop application" ON)
option(CRUCIBLE_SANITIZERS "Enable address/undefined sanitizers" OFF)
include(CTest)
find_package(Threads REQUIRED)
add_library(crucible_core core/flow.cpp core/medium.cpp core/session.cpp core/transport.cpp core/walls.cpp)
target_include_directories(crucible_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(crucible_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(crucible_core PRIVATE /W4)
else()
  target_compile_options(crucible_core PRIVATE -Wall -Wextra -Wpedantic)
  if(CRUCIBLE_SANITIZERS)
    target_compile_options(crucible_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(crucible_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(crucible_run app/run.cpp)
target_link_libraries(crucible_run PRIVATE crucible_core)
if(BUILD_TESTING)
  add_executable(crucible_tests tests/core_tests.cpp)
  target_link_libraries(crucible_tests PRIVATE crucible_core)
  add_test(NAME core_verification COMMAND crucible_tests)
  add_test(NAME core_slow_verification COMMAND crucible_tests --slow)
  set_tests_properties(core_slow_verification PROPERTIES LABELS slow)
  add_executable(crucible_convergence tests/nozzle_convergence.cpp)
  target_link_libraries(crucible_convergence PRIVATE crucible_core)
  add_executable(crucible_low_mach_study tests/low_mach_study.cpp)
  target_link_libraries(crucible_low_mach_study PRIVATE crucible_core)
  add_executable(crucible_low_mach_stability tests/low_mach_stability.cpp)
  target_link_libraries(crucible_low_mach_stability PRIVATE crucible_core)
endif()
if(CRUCIBLE_BUILD_APP)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets OpenGLWidgets)
  find_package(VTK 9.3 REQUIRED COMPONENTS CommonCore CommonDataModel
    FiltersCore RenderingCore RenderingOpenGL2 RenderingAnnotation
    InteractionStyle GUISupportQt)
  qt_standard_project_setup()
  qt_add_executable(crucible MACOSX_BUNDLE WIN32 app/main.cpp app/window.cpp)
  target_link_libraries(crucible PRIVATE crucible_core Qt6::Widgets Qt6::OpenGLWidgets ${VTK_LIBRARIES})
  vtk_module_autoinit(TARGETS crucible MODULES ${VTK_LIBRARIES})
  set_target_properties(crucible PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME CRUCIBLE
    MACOSX_BUNDLE_GUI_IDENTIFIER org.crucible.sandbox)
endif()
# Parked architecture spike (docs/parked/MHD_SPIKE.md): throwaway MHD probe on the RZ layout.
add_executable(crucible_mhd_spike spike/mhd_rz.cpp)
target_link_libraries(crucible_mhd_spike PRIVATE crucible_core)
# Real-gas thermochemistry adapter (Cantera, built from source; TECHNICAL_PLAN adapters/).
option(CRUCIBLE_WITH_CANTERA "Build the Cantera thermochemistry adapter" ON)
set(CANTERA_ROOT "$ENV{HOME}/src/cantera" CACHE PATH "Cantera source tree with a scons build")
if(CRUCIBLE_WITH_CANTERA)
  find_library(ACCELERATE_FRAMEWORK Accelerate)
  find_path(BOOST_INCLUDE boost/shared_ptr.hpp HINTS /opt/homebrew/include /usr/local/include)
  add_library(crucible_thermo adapters/thermo.cpp adapters/reaction.cpp adapters/reacting_flow.cpp)
  target_include_directories(crucible_thermo PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}
    PRIVATE ${CANTERA_ROOT}/include ${CANTERA_ROOT}/include/cantera/ext ${BOOST_INCLUDE})
  target_compile_definitions(crucible_thermo PRIVATE CRUCIBLE_CANTERA_DATA="${CANTERA_ROOT}/data")
  target_link_libraries(crucible_thermo PUBLIC crucible_core PRIVATE ${CANTERA_ROOT}/build/lib/libcantera.a
    ${ACCELERATE_FRAMEWORK} Threads::Threads)
  if(BUILD_TESTING)
    add_executable(crucible_thermo_tests tests/thermo_tests.cpp)
    target_link_libraries(crucible_thermo_tests PRIVATE crucible_thermo)
    target_compile_definitions(crucible_thermo_tests PRIVATE CRUCIBLE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME thermo_cea_verification COMMAND crucible_thermo_tests)
    # The reference reactor network is Cantera's own, so this test sees Cantera directly.
    add_executable(crucible_reaction_tests tests/reaction_tests.cpp)
    target_include_directories(crucible_reaction_tests PRIVATE ${CANTERA_ROOT}/include ${BOOST_INCLUDE})
    target_link_libraries(crucible_reaction_tests PRIVATE crucible_thermo
      ${CANTERA_ROOT}/build/lib/libcantera.a ${ACCELERATE_FRAMEWORK} Threads::Threads)
    add_test(NAME reaction_verification COMMAND crucible_reaction_tests)
    add_executable(crucible_transport_tests tests/transport_tests.cpp)
    target_link_libraries(crucible_transport_tests PRIVATE crucible_thermo)
    add_test(NAME transport_verification COMMAND crucible_transport_tests)
    add_executable(crucible_turbulence_tests tests/turbulence_tests.cpp)
    target_link_libraries(crucible_turbulence_tests PRIVATE crucible_thermo)
    target_compile_definitions(crucible_turbulence_tests PRIVATE CRUCIBLE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME turbulence_verification COMMAND crucible_turbulence_tests)
    add_test(NAME turbulence_pipe_verification COMMAND crucible_turbulence_tests --pipe)
    set_tests_properties(turbulence_pipe_verification PROPERTIES LABELS slow)
    add_executable(crucible_mixture_tests tests/mixture_tests.cpp)
    target_link_libraries(crucible_mixture_tests PRIVATE crucible_thermo)
    add_test(NAME mixture_verification COMMAND crucible_mixture_tests)
    add_executable(crucible_chemistry_cost tests/chemistry_cost.cpp)
    target_include_directories(crucible_chemistry_cost PRIVATE ${CANTERA_ROOT}/include ${BOOST_INCLUDE})
    target_link_libraries(crucible_chemistry_cost PRIVATE crucible_thermo
      ${CANTERA_ROOT}/build/lib/libcantera.a ${ACCELERATE_FRAMEWORK} Threads::Threads)
    add_executable(crucible_step_cost tests/step_cost.cpp)
    target_link_libraries(crucible_step_cost PRIVATE crucible_thermo)
    add_executable(crucible_chamber_study tests/chamber_study.cpp)
    target_include_directories(crucible_chamber_study PRIVATE ${CANTERA_ROOT}/include ${BOOST_INCLUDE})
    target_link_libraries(crucible_chamber_study PRIVATE crucible_thermo
      ${CANTERA_ROOT}/build/lib/libcantera.a ${ACCELERATE_FRAMEWORK} Threads::Threads)
    add_executable(crucible_flame_study tests/flame_study.cpp)
    target_include_directories(crucible_flame_study PRIVATE ${CANTERA_ROOT}/include ${BOOST_INCLUDE})
    target_link_libraries(crucible_flame_study PRIVATE crucible_thermo
      ${CANTERA_ROOT}/build/lib/libcantera.a ${ACCELERATE_FRAMEWORK} Threads::Threads)
    add_executable(crucible_detonation_study tests/detonation_study.cpp)
    target_include_directories(crucible_detonation_study PRIVATE ${CANTERA_ROOT}/include ${BOOST_INCLUDE})
    target_link_libraries(crucible_detonation_study PRIVATE crucible_thermo
      ${CANTERA_ROOT}/build/lib/libcantera.a ${ACCELERATE_FRAMEWORK} Threads::Threads)
  endif()
endif()
